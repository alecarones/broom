 16def pilc(config: Configs, input_alms: SimpleNamespace, compsep_run: Dict[str, Any], **kwargs) -> Union[SimpleNamespace, None]:
 17    """
 18    Perform Polarization Internal Linear Combination (PILC) on input alm coefficients.
 19
 20    Parameters
 21    ----------
 22        config : Configs
 23            Configuration object containing global settings. It includes:
 24            - lmax : int, maximum multipole for the component separation.
 25            - nside : int, HEALPix resolution parameter for compsep products.
 26            - fwhm_out : float, full width at half maximum of the output beam in arcminutes.
 27            - pixel_window_out : bool, whether to apply a pixel window to the output maps.
 28            - field_out : str, desired output fields (e.g., "E", "B", "QU", "QU_E", "QU_B").
 29            - save_compsep_products : bool, whether to save component separation products.
 30            - return_compsep_products : bool, whether to return component separation products.
 31            - path_outputs : str, path to save the output files.
 32            - units : str, units of the output maps (e.g., "uK_CMB"). Used to compute components SEDs, if needed.
 33            - bandpass_integrate : bool, whether inputs are bandpass-integrated. Used to compute components SEDs, if needed. 
 34
 35        input_alms : SimpleNamespace
 36            SimpleNamespace object containing input spherical harmonic coefficients (alms). 
 37            Each attribute has shape (n_channels, (n_fields), n_alms), where n_fields depend on the fields requested in config.field_out. 
 38            n_fields can be None if only E and B are requested, or 2 if both are requested to be included in P component separation.
 39
 40        compsep_run : dict
 41            Dictionary specifying component separation parameters. It includes:
 42            - "method": Method to use for component separation (e.g., "pilc", "cpilc", "prilc", "cprilc", ..)
 43            - "domain": Domain of the component separation (e.g., "pixel", "needlet").
 44            - "needlet_config": Configuration for needlet parameters (if applicable).
 45            - "ilc_bias": Parameter setting the residual percentage ILC bias (default is 0. and covariance will be computed over the whole sky).
 46            - "reduce_ilc_bias": Boolean. If True, it implements a procedure to attenuate ILC bias.
 47            - "save_needlets": boolean. Whether to save adopted needlet bands in the output directory. 
 48                If not provided, it is set to the value of config.save_compsep_products. Default: True.
 49            - "save_weights": boolean. Whether to save compsep weights maps in the output directory.
 50                If not provided, it is set to the value of config.save_compsep_products. Default: True.
 51            - "mask": (optional) HEALPix mask which will exclude (unobserved) regions from covariance computation. 
 52                It must be a 1D array with shape (12 * nside**2,). If non-binary, it will be used to weigh pixel in covariance computation.
 53            - "cov_noise_debias": (float, list) Noise covariance debiasing factor. 
 54                If different from 0. it will debias the covariance matrix by a factor cov_noise_debias * noise_covariance.
 55                It must be a list with the same length as the number of needlet scales if domain is "needlet", otherwise a single float.
 56            - "special_nls": (list) List of needlet scales where component deprojection is applied in c_pilc.
 57            - "constraints": Dictionary to be used if method is "cpilc", "cprilc", "c_pilc" or "c_prilc". It must contain:
 58                - "components": list of strings with components to be deprojected in cPILC. It can be a list of lists if domain is "needlet".
 59                    Each list will be associated to a needlet band in order.
 60            Optionally can include:
 61                - "beta_d": (float, list): dust spectral index to compute dust moments. If list, a different beta_d is used for each needlet band. Default: 1.54.
 62                - "T_d": (float, list): dust temperature to compute dust moments. If list, a different T_d is used for each needlet band. Default: 19.6.
 63                - "beta_s": (float, list): synchrotron spectral index to compute synchrotron moments. If list, a different beta_s is used for each needlet band. Default: -3.
 64                - "deprojection": (float, list): deprojection factor for each component. If list, a different deprojection factor is used for each component.
 65                    It can be a list of lists if domain is "needlet". Each list will be associated to a needlet band in order. Default: 0. (i.e. full deprojection) for each component and each needlet band.
 66
 67        **kwargs : dict
 68            Additional keyword arguments forwarded to hp.map2alm.
 69
 70    Returns
 71    -------
 72        SimpleNamespace or None
 73            Output maps if config.return_compsep_products is True, otherwise None.
 74    """
 75
 76    if config.field_out not in ["E", "B", "QU", "EB", "QU_E", "QU_B"]:
 77        raise ValueError("Invalid field_out for P(R)ILC. It must be E, B, QU, EB, QU_E, or QU_B.")
 78
 79    if hasattr(input_alms, "total"):
 80        compsep_run["from_splits"] = False
 81    elif hasattr(input_alms, "total_split1") and hasattr(input_alms, "total_split2"):
 82        compsep_run["from_splits"] = True
 83    else:
 84        raise ValueError("The input_alms object must have either 'total' attribute or both 'total_split1' and 'total_split2' attributes.")
 85
 86    if compsep_run["method"] in ["cpilc", "c_pilc", "cprilc", "c_prilc"]:
 87        compsep_run = _standardize_cilc(compsep_run, config.lmax)
 88
 89    input_attrs = obj_to_array(input_alms, return_attributes=True)
 90
 91    if not compsep_run["from_splits"]:
 92        if np.any(np.array(compsep_run["cov_noise_debias"]) != 0.):
 93            if not ("load_noise_covariance" in compsep_run and compsep_run["load_noise_covariance"]):
 94                if not hasattr(input_alms, "noise"):
 95                    raise ValueError("The input_alms object must have 'noise'' attribute for debiasing the covariance.")
 96                compsep_run["noise_idx"] = input_attrs.index('noise')
 97
 98    output_maps = _pilc(config, obj_to_array(input_alms), compsep_run, **kwargs)
 99    
100    outputs = array_to_obj(output_maps, input_attrs)
101
102    del output_maps
103
104    return outputs, compsep_run

225def _pilc_needlet_j(config: Configs, input_alms: np.ndarray, compsep_run: Dict[str, Any], b_ell: np.ndarray, nl_scale: int) -> np.ndarray:
226    """
227    Compute the needlet-scale P(R)ILC component separated maps.
228
229    Parameters
230    ----------
231        config : Configs
232            Configuration object. See 'pilc' function for details.
233        input_alms : np.ndarray
234            Input alm array. It can have shape:
235            - (n_channels, 2, n_alms, n_comps) for EB inputs

238            Component separation configuration. See 'pilc' function for details.

241        nl_scale : int
242            Current needlet scale index.
243    
244    Returns
245    -------
246        np.ndarray
247            Output alm array at the requested needlet scale.
248    """    
249    nside_, lmax_ = config.nside, config.lmax
250
251    good_channels_nl = _get_good_channels_nl(config, b_ell)
252
253    input_maps_nl = np.zeros((good_channels_nl.shape[0], 2, 12 * nside_**2, input_alms.shape[-1]))
254    for n, channel in enumerate(good_channels_nl):
255        input_alms_j = np.zeros((2, hp.Alm.getsize(lmax_), input_alms.shape[-1]), dtype=complex)

257            for k in range(2):

417        idx = scale_idx if scale_idx is not None else slice(None)
418        compsep_run["A"] = _get_SEDs(
419            freqs,
420            compsep_run["constraints"]["components"][idx],
421            beta_d=compsep_run["constraints"]["beta_d"][idx],
422            T_d=compsep_run["constraints"]["T_d"][idx],
423            beta_s=compsep_run["constraints"]["beta_s"][idx],
424            units=config.units,
425            bandwidths=bandwidths
426        )
427        compsep_run["e"] = np.array(compsep_run["constraints"]["deprojection"][idx])
428
429    if compsep_run["method"] in ["cpilc", "cprilc"]:
430        _assign_component_constraints(nl_scale)
431    elif compsep_run["method"] in ["c_pilc", "c_prilc"] and nl_scale in compsep_run.get("special_nls", []):
432        scale_idx = np.where(np.array(compsep_run["special_nls"]) == nl_scale)[0][0]        
433        _assign_component_constraints(scale_idx)
434
435    if compsep_run["from_splits"]:
436        if "prilc" in compsep_run["method"]:
437            cov = get_prilc_cov(input_maps[...,0], config.lmax, compsep_run, config.fwhm_out, b_ell, input_maps_2=input_maps[...,1])
438        else:
439            cov = get_pilc_cov(input_maps[...,0], config.lmax, compsep_run, config.fwhm_out, b_ell, input_maps_2=input_maps[...,1])
440    else:
441        if "prilc" in compsep_run["method"]:
442            cov = get_prilc_cov(input_maps[...,0], config.lmax, compsep_run, config.fwhm_out, b_ell)
443        else:
444            cov = get_pilc_cov(input_maps[...,0], config.lmax, compsep_run, config.fwhm_out, b_ell)
445        noise_debias = compsep_run["cov_noise_debias"] if compsep_run["domain"] == "pixel" else compsep_run["cov_noise_debias"][nl_scale]
446        if noise_debias != 0.:
447            if not ("load_noise_covariance" in compsep_run and compsep_run["load_noise_covariance"]):
448                if "prilc" in compsep_run["method"]:
449                    cov_n = get_prilc_cov(input_maps[...,compsep_run["noise_idx"]], config.lmax, compsep_run, config.fwhm_out, b_ell)
450                else:
451                    cov_n = get_pilc_cov(input_maps[...,compsep_run["noise_idx"]], config.lmax, compsep_run, config.fwhm_out, b_ell)
452            else:
453                path_noicov = _get_full_path_nuiscov(config, compsep_run)
454                cov_n = (load_nuiscov(config, path_noicov, compsep_run,
455                                    hp.npix2nside(input_maps.shape[-2]), "noise", nl_scale=nl_scale)).T
456            cov = cov - noise_debias * cov_n
457            del cov_n
458
459    inv_cov = get_inv_cov(cov)
460    del cov
461    if "prilc" in compsep_run["method"]:
462        w_pilc = get_prilc_weights(A_out, inv_cov, input_maps.shape, compsep_run)
463    else:
464        w_pilc = get_pilc_weights(A_out, inv_cov, input_maps.shape, compsep_run)
465    del inv_cov
466
467    if compsep_run["save_weights"]:
468        #if 'path_out' not in compsep_run:
469        compsep_run["path_out"] = _get_full_path_out(config, compsep_run)
470        save_ilc_weights(config, w_pilc, compsep_run,
471                         hp.npix2nside(input_maps.shape[-2]), nl_scale=nl_scale)
472        
473    compsep_run.pop("A", None)
474    compsep_run.pop("e", None)
475
476    if compsep_run["ilc_bias"] == 0.:
477        if w_pilc.ndim==1:
478            output_maps = np.einsum('i,ifjk->fjk', w_pilc, input_maps)
479        elif w_pilc.ndim==2:
480            Q = np.einsum('i,ijk->jk', w_pilc[0], input_maps[:, 0])
481            U = np.einsum('i,ijk->jk', w_pilc[1], input_maps[:, 0])
482            Q -= np.einsum('i,ijk->jk', w_pilc[1], input_maps[:, 1])
483            U += np.einsum('i,ijk->jk', w_pilc[0], input_maps[:, 1])
484            output_maps = np.stack([Q, U])
485    else:
486        if w_pilc.ndim==2:
487            output_maps = np.einsum('ij,ifjk->fjk', w_pilc, input_maps)
488        elif w_pilc.ndim==3:
489            Q = np.einsum('ij,ijk->jk', w_pilc[0], input_maps[:, 0])
490            U = np.einsum('ij,ijk->jk', w_pilc[1], input_maps[:, 0])
491            Q -= np.einsum('ij,ijk->jk', w_pilc[1], input_maps[:, 1])
492            U += np.einsum('ij,ijk->jk', w_pilc[0], input_maps[:, 1])
493            output_maps = np.stack([Q, U])
494            
495    return output_maps

497def get_prilc_weights(
498    A_out: np.ndarray,
499    inv_cov: np.ndarray,
500    input_shapes: Tuple[int, ...],
501    compsep_run: Dict
502) -> np.ndarray:
503    """
504    Compute real weights for the Polarization Internal Linear Combination (PRILC) method,
505    with or without constraints to deproject additional components.
506
507    Parameters:
508    ----------
509        A_out: np.ndarray
510            Spectral energy distribution (SED) of the component to be reconstructed (e.g., CMB). Shape (n_channels,).
511        inv_cov: np.ndarray
512            Inverse covariance matrix of input polarization maps:
513            - If ilc_bias=0: shape (n_channels, n_channels)
514            - If ilc_bias!=0: shape (n_channels, n_channels, n_pixels)
515        input_shapes: Tuple[int, ...]
516            Shape of the input map (n_channels, 2, n_pixels, n_comps) or equivalent, used to determine spatial size and channels.
517        compsep_run: Dict
518            Dictionary with component separation settings. See 'pilc' function for details.
519
520    Returns
521    -------
522        w_ilc: np.ndarray
523            ILC weights, shaped:
524                - (n_channels,) if ilc_bias = 0
525                - (n_channels, n_pixels) if ilc_bias != 0
526    """
527
528    if "A" in compsep_run:
529        # Add component constraint at the top of the A matrix and 'e' constraint vector
530        compsep_run["A"] = np.vstack((A_out, compsep_run["A"]))
531        compsep_run["e"] = np.insert(compsep_run["e"], 0, 1.)
532
533        if compsep_run["ilc_bias"] == 0.:
534            # Spatially-invariant weights (simpler case)
535            inv_ACA = np.linalg.inv(
536                np.einsum("zi,il->zl", compsep_run["A"], np.einsum("ij,lj->il", inv_cov, compsep_run["A"]))
537            )
538            w_ilc = np.einsum(
539                "l,lj->j",
540                compsep_run["e"],
541                np.einsum("lz,zj->lj", inv_ACA, np.einsum("zi,ij->zj", compsep_run["A"], inv_cov))
542            )
543            del inv_ACA
544        else:
545            # Spatially-varying weights
546            inv_ACA = np.linalg.inv(
547                np.einsum("zi,ilk->zlk", compsep_run["A"], np.einsum("ijk,lj->ilk", inv_cov, compsep_run["A"])).T).T
548
549            w_ilc=np.zeros((input_shapes[0],input_shapes[-2]))
550            w_ = np.einsum(
551                "l,ljk->jk",
552                compsep_run["e"],
553                np.einsum("lzk,zjk->ljk", inv_ACA, np.einsum("zi,ijk->zjk", compsep_run["A"], inv_cov))
554            )
555
556            for i in range(input_shapes[0]):
557                #if "mask" in compsep_run:
558                #    w_ilc[i] = w_[i]
559                #else:
560                w_ilc[i]=hp.ud_grade(w_[i],hp.npix2nside(input_shapes[-2]))
561            del w_, inv_ACA
562    else:
563        if compsep_run["ilc_bias"] == 0.:
564            w_ilc = (A_out.T @ inv_cov) / (A_out.T @ inv_cov @ A_out) 
565        else:
566            w_ilc=np.zeros((input_shapes[0],input_shapes[-2]))
567            AT_invC = np.einsum('j,ijk->ik', A_out, inv_cov) # np.sum(inv_cov,axis=1)
568            AT_invC_A = np.einsum('j,ijk, i->k', A_out, inv_cov, A_out) #np.sum(inv_cov,axis=(0,1))
569            for i in range(input_shapes[0]):
570                #if "mask" in compsep_run:
571                #    w_ilc[i] = AT_invC[i]/AT_invC_A
572                #else:
573                w_ilc[i]=hp.ud_grade(AT_invC[i]/AT_invC_A,hp.npix2nside(input_shapes[-2]))
574    return w_ilc

576def get_pilc_weights(
577    A_out: np.ndarray,
578    inv_cov: np.ndarray,
579    input_shapes: Tuple[int, ...],
580    compsep_run: Dict
581) -> np.ndarray:
582    """
583    Compute weights for the Polarization Internal Linear Combination (PILC) method,
584    with or without constraints to deproject additional components.
585
586    Parameters:
587    ----------
588        A_out: np.ndarray
589            Spectral energy distribution (SED) of the component to be reconstructed (e.g., CMB). Shape (n_channels,).
590        inv_cov: np.ndarray
591            Inverse covariance matrix of input polarization maps:
592            - If ilc_bias=0: shape (n_channels, n_channels)
593            - If ilc_bias!=0: shape (n_channels, n_channels, n_pixels)
594        input_shapes: Tuple[int, ...]
595            Shape of the input map (n_channels, 2, n_pixels, n_comps) or equivalent, used to determine spatial size and channels.
596        compsep_run: Dict
597            Dictionary with component separation settings. See 'pilc' function for details.
598
599    Returns
600    -------
601        w_ilc: np.ndarray
602            ILC weights, shaped:
603                - (n_channels,) if ilc_bias = 0
604                - (n_channels, n_pixels) if ilc_bias != 0
605    """
606
607    if "A" in compsep_run:
608        compsep_run["A"] = np.vstack((A_out, compsep_run["A"]))
609        compsep_run["e"] = np.insert(compsep_run["e"], 0, 1.)
610        if compsep_run["ilc_bias"] == 0.:
611            Mp = np.einsum('zi,ij,lj->zl', compsep_run["A"], inv_cov[:input_shapes[0],:input_shapes[0]], compsep_run["A"])
612            Mm = np.einsum('zi,ij,lj->zl', compsep_run["A"], inv_cov[:input_shapes[0],input_shapes[0]:], compsep_run["A"])
613            inv_M = np.linalg.inv(Mp + np.einsum('zl,lj,ji->zi', Mm, np.linalg.inv(Mp), Mm))
614            lr = 2. * np.einsum('zl,l->z', inv_M, compsep_run["e"])
615            li = 2. * np.einsum('zl,li,ij,j->z', np.linalg.inv(Mp), Mm, inv_M, compsep_run["e"])
616            del inv_M, Mp, Mm
617
618            w_ilc = np.zeros((2, input_shapes[0]))
619            w_ilc[0] = 0.5 * np.einsum('ij,zj,z->i', inv_cov[:input_shapes[0],:input_shapes[0]], compsep_run["A"], lr)
620            w_ilc[0] += 0.5 * np.einsum('ij,zj,z->i', inv_cov[:input_shapes[0],input_shapes[0]:], compsep_run["A"], li)
621            w_ilc[1] = 0.5 * np.einsum('ij,zj,z->i', inv_cov[input_shapes[0]:,:input_shapes[0]], compsep_run["A"], lr)
622            w_ilc[1] += 0.5 * np.einsum('ij,zj,z->i', inv_cov[input_shapes[0]:,input_shapes[0]:], compsep_run["A"], li)
623            
624        else:
625            Mp = np.einsum('zi,ijk,lj->zlk', compsep_run["A"], inv_cov[:input_shapes[0],:input_shapes[0]], compsep_run["A"])
626            Mm = np.einsum('zi,ijk,lj->zlk', compsep_run["A"], inv_cov[:input_shapes[0],input_shapes[0]:], compsep_run["A"])
627            comb_M = np.transpose(
628                Mp, (2,0,1)) + np.einsum('zlk,klj,jik->kzi', Mm, np.linalg.inv(np.transpose(
629                Mp, (2,0,1))), Mm)
630            inv_M = np.transpose(np.linalg.inv(comb_M), (1,2,0))
631            del comb_M
632            lr = 2. * np.einsum('zlk,l->zk', inv_M, compsep_run["e"])
633            li = 2. * np.einsum('kzl,lik,ijk,j->zk', np.linalg.inv(np.transpose(Mp, (2,0,1))), Mm, inv_M, compsep_run["e"])
634            del inv_M, Mp, Mm
635
636            w_ilc = np.zeros((2, input_shapes[0], input_shapes[-2]))
637            w_ = 0.5 * np.einsum('ijk,zj,zk->ik', inv_cov[:input_shapes[0],:input_shapes[0]], compsep_run["A"], lr)
638            w_ += 0.5 * np.einsum('ijk,zj,zk->ik', inv_cov[:input_shapes[0],input_shapes[0]:], compsep_run["A"], li)
639            for i in range(input_shapes[0]):
640                w_ilc[0,i]=hp.ud_grade(w_[i],hp.npix2nside(input_shapes[-2]))
641            
642            w_ = 0.5 * np.einsum('ijk,zj,zk->ik', inv_cov[input_shapes[0]:,:input_shapes[0]], compsep_run["A"], lr)
643            w_ += 0.5 * np.einsum('ijk,zj,zk->ik', inv_cov[input_shapes[0]:,input_shapes[0]:], compsep_run["A"], li)
644            del lr, li
645            for i in range(input_shapes[0]):
646                w_ilc[1,i]=hp.ud_grade(w_[i],hp.npix2nside(input_shapes[-2]))
647            del w_
648            
649    else:
650        if compsep_run["ilc_bias"] == 0.:
651            w_ilc = np.zeros((2, input_shapes[0]))
652
653            Sp = A_out.T @ (inv_cov[:input_shapes[0],:input_shapes[0]]) @ A_out
654            Sm = A_out.T @ (inv_cov[:input_shapes[0],input_shapes[0]:]) @ A_out
655
656            lr = (2. * Sp) / (Sp**2 - Sm**2)
657            li = - (2. * Sm) / (Sp**2 - Sm**2)
658            del Sp, Sm
659
660            w_ilc[0] = 0.5 * (lr * (inv_cov[:input_shapes[0],:input_shapes[0]] @ A_out) +
661                                li * (inv_cov[:input_shapes[0],input_shapes[0]:] @ A_out))
662            w_ilc[1] = 0.5 * (lr * (inv_cov[input_shapes[0]:,:input_shapes[0]] @ A_out) +
663                                li * (inv_cov[input_shapes[0]:,input_shapes[0]:] @ A_out))
664
665        else:
666            w_ilc = np.zeros((2, input_shapes[0], input_shapes[-2]))
667            
668            Sp = np.einsum('i,ijk,j->k', A_out, inv_cov[:input_shapes[0],:input_shapes[0]], A_out)
669            Sm = np.einsum('i,ijk,j->k', A_out, inv_cov[:input_shapes[0],input_shapes[0]:], A_out)
670
671            lr = (2. * Sp) / (Sp**2 - Sm**2)
672            li = - (2. * Sm) / (Sp**2 - Sm**2)
673
674            del Sp, Sm
675
676            w_ = 0.5 * np.einsum('k,ijk,j->ik', lr, inv_cov[:input_shapes[0],:input_shapes[0]], A_out)
677            w_ += 0.5 * np.einsum('k,ijk,j->ik', li, inv_cov[:input_shapes[0],input_shapes[0]:], A_out)
678            for i in range(input_shapes[0]):
679                w_ilc[0,i]=hp.ud_grade(w_[i],hp.npix2nside(input_shapes[-2]))
680
681            w_ = 0.5 * np.einsum('k,ijk,j->ik', lr, inv_cov[input_shapes[0]:,:input_shapes[0]], A_out)
682            w_ += 0.5 * np.einsum('k,ijk,j->ik', li, inv_cov[input_shapes[0]:,input_shapes[0]:], A_out)
683            del lr, li
684            for i in range(input_shapes[0]):
685                w_ilc[1,i]=hp.ud_grade(w_[i],hp.npix2nside(input_shapes[-2]))
686        
687    return w_ilc
