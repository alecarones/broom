  19    Perform Internal Linear Combination (ILC) component separation (all its variants) on CMB scalar fields.
  20
  21    Parameters
  22    ----------
  23        config : Configs
  24            Configuration object containing global settings. It includes:
  25            - lmax : int, maximum multipole for the component separation.
  26            - nside : int, HEALPix resolution parameter for compsep products.
  27            - fwhm_out : float, full width at half maximum of the output beam in arcminutes.
  28            - pixel_window_out : bool, whether to apply a pixel window to the output maps.
  29            - field_out : str, desired output fields (e.g., "T", "E", "B", "QU", "TQU", "QU_E", "QU_B").

  31            - return_compsep_products : bool, whether to return component separation products.
  32            - path_outputs : str, path to save the output files.
  33            - units : str, units of the output maps (e.g., "uK_CMB"). Used to compute components SEDs, if needed.
  34            - bandpass_integrate : bool, whether inputs are bandpass-integrated. Used to compute components SEDs, if needed. 
  35        input_alms : SimpleNamespace
  36            SimpleNamespace object containing input spherical harmonic coefficients (alms). 
  37            Each attribute has shape (n_channels, (n_fields), n_alms), where n_fields depend on the fields requested in config.field_out.
  38        compsep_run : dict
  39            Dictionary specifying component separation parameters. It includes:
  40            - "method": Method to use for component separation (e.g., "ilc", "cilc", "mc_ilc").
  41            - "domain": Domain of the component separation (e.g., "pixel", "needlet").
  42            - "needlet_config": Dictionary containing needlet settings:
  43                - "needlet_windows": Type of needlet windows ('cosine', 'standard', 'mexican').
  44                - "ell_peaks": List of integers defining multipoles of peaks for needlet bands (required for 'cosine').
  45                - "width": Width of the needlet windows (required for 'standard' and 'mexican').
  46                - "merging_needlets": Integer or list of integers defining ranges of needlets to be merged.
  47            - "ilc_bias": Parameter setting the residual percentage ILC bias (default is 0. and covariance will be computed over the whole sky).
  48            - "reduce_ilc_bias": Boolean. If True, it implements a procedure to attenuate ILC bias.
  49            - "b_squared": Boolean indicating if the needlet windows should be squared (if domain is "needlet").
  50            - "adapt_nside": Adapt HEALPix resolution of needlet maps to sampled multipole range. Deafult: False.
  51            - "save_needlets": boolean. Whether to save adopted needlet bands in the output directory. 
  52                If not provided, it is set to the value of config.save_compsep_products. Default: True.
  53            - "save_weights": boolean. Whether to save compsep weights maps in the output directory.
  54                If not provided, it is set to the value of config.save_compsep_products. Default: True.
  55            - "mask": (optional) HEALPix mask which will exclude (unobserved) regions from covariance computation. 
  56                It must be a 1D array with shape (12 * nside**2,). If non-binary, it will be used to weigh pixel in covariance computation.
  57            - "path_patches": (str) Path to .npy file containing pre-defined patches for MC-ILC. If not provided, patches will be defined on the fly. Default: None.
  58            - "n_patches": (int) Number of patches to use in MC-ILC. Default: 50.
  59            - "mc_type": (str), needed if method is "mcilc" or "mc_ilc".
  60                Type of MC-ILC to use. Options: "cea_ideal", "rp_ideal", "cea_real", "rp_real". Default: "cea_real".
  61            - "cov_noise_debias": (float, list) Noise covariance debiasing factor. 
  62                If different from 0. it will debias the covariance matrix by a factor cov_noise_debias * noise_covariance.

  64            - "special_nls": (list) List of needlet scales where moment deprojection is applied in c_ilc and clustering in mc_ilc.
  65            - "constraints": Dictionary to be used if method is "cilc" or "c_ilc". It must contain:

  72                - "deprojection": (float, list): deprojection factor for each moment. If list, a different deprojection factor is used for each moment.
  73                    It can be a list of lists if domain is "needlet". Each list will be associated to a needlet band in order. Default: 0. (i.e. full deprojection) for each moment and each needlet band.

 130    outputs = array_to_obj(output_maps, input_attrs)

 136def _ilc_post_processing(
 137    config: Configs,
 138    output_maps: np.ndarray,
 139    compsep_run: Dict,
 140    **kwargs: Any
 141) -> np.ndarray:
 142    """
 143    Post-process the ILC output maps, including polarization conversions and masking.
 144
 145    Parameters
 146    ----------
 147        config : Configs
 148            Configuration object containing output settings. See 'ilc' for details.
 149        output_maps : np.ndarray
 150            Output maps from the ILC process. 
 151            Shape is (n_fields, npix, n_components) where n_fields depends on the requested output field in config.field_out.
 152        compsep_run : dict
 153            Dictionary specifying component separation parameters. It requires:
 154            - "mask": (optional) HEALPix mask which will exclude (unobserved) regions from the output maps.
 155        **kwargs : dict
 156            Additional keyword arguments forwarded to hp.map2alm.
 157
 158    Returns
 159    -------
 160        np.ndarray
 161            Post-processed output maps.
 162    """
 163    # Handle EB to QU conversion for 3D arrays with appropriate fields
 164    if output_maps.ndim == 3 and (
 165        (output_maps.shape[0] == 2 and config.field_out == "QU") or
 166        (output_maps.shape[0] == 3 and config.field_out == "TQU")
 167    ):
 168        outputs = np.zeros_like(output_maps)
 169        for c in range(output_maps.shape[-1]):
 170            outputs[:,:,c] = _EB_to_QU(output_maps[:,:,c],config.lmax, **kwargs)
 171        if "mask" in compsep_run:
 172            outputs[:,compsep_run["mask"] == 0.,:] = 0.
 173        return outputs
 174    # Handle QU_E and QU_B fields for 2D arrays
 175    elif (output_maps.ndim==2) and (config.field_out in ["QU_E", "QU_B"]):
 176        output = np.zeros((2, output_maps.shape[0], output_maps.shape[-1]))
 177        for c in range(output_maps.shape[-1]):
 178            if config.field_out == "QU_E":
 179                output[:,:,c] = _E_to_QU(output_maps[:,c],config.lmax, **kwargs)
 180            elif config.field_out=="QU_B":
 181                output[:,:,c] = _B_to_QU(output_maps[:,c],config.lmax, **kwargs)
 182        if "mask" in compsep_run:
 183            output[:,compsep_run["mask"] == 0.,:] = 0.
 184        return output
 185    # For other cases, apply mask if present
 186    else:
 187        if "mask" in compsep_run:
 188            if output_maps.ndim == 3:
 189                output_maps[:, compsep_run["mask"] == 0., :] = 0.
 190            elif output_maps.ndim == 2:
 191                output_maps[compsep_run["mask"] == 0.,:] = 0.
 192        return output_maps

 194def _ilc(
 195    config: Configs,
 196    input_alms: np.ndarray,
 197    compsep_run: Dict,
 198    **kwargs: Any
 199) -> np.ndarray:
 200    """
 201    Perform ILC on input alm arrays, either scalar or multi-field.
 202
 203    Parameters
 204    ----------
 205        config : Configs
 206            Configuration object containing global settings. See 'ilc' for details.
 207        input_alms : np.ndarray
 208            Input spherical harmonic coefficients. 
 209            It must have shape (n_channels, n_fields, n_alms, n_components) for multifield or 
 210            (n_channels, n_alms, n_components) for scalar fields.
 211        compsep_run : dict
 212            Component separation configuration dictionary. See 'ilc' for details.
 213        **kwargs : dict
 214            Additional keyword arguments for hp.map2alm.
 215
 216    Returns
 217    -------
 218        np.ndarray
 219            Output separated maps.
 220    """
 221    # Determine fields based on input shape and desired output
 222    fields_ilc = get_fields_from_alms(input_alms, config.field_out)
 223
 224    # Allocate output array and run ILC compsep per field if needed
 225    if input_alms.ndim == 4:
 226        output_maps = np.zeros((input_alms.shape[1], 12 * config.nside**2, input_alms.shape[3]))
 227        for i in range(input_alms.shape[1]):
 228            compsep_run["field"] = fields_ilc[i]
 229            if compsep_run["method"] in ["mcilc", "mc_ilc", "mc_cilc"]:
 230                if "path_patches" not in compsep_run:
 231                    input_fgds_alms = np.zeros_like(input_alms[:, i, :, 0]) if "real" in compsep_run["mc_type"] else input_alms[:, i, :, compsep_run["fgds_idx"]]
 232                    compsep_run["tracers"] = initialize_scalar_tracers(
 233                        config, input_fgds_alms, compsep_run, field=compsep_run["field"], **kwargs
 234                    )
 235                    del input_fgds_alms
 236                else:
 237                    compsep_run["field_idx"] = i
 238            output_maps[i] = _ilc_scalar(config, input_alms[:, i, :, :], compsep_run, **kwargs)
 239
 240    elif input_alms.ndim == 3:
 241        compsep_run["field"] = fields_ilc[0]
 242        if compsep_run["method"] in ["mcilc", "mc_ilc", "mc_cilc"]:
 243            if "path_patches" not in compsep_run:
 244                input_fgds_alms = np.zeros_like(input_alms[...,0]) if "real" in compsep_run["mc_type"] else input_alms[...,compsep_run["fgds_idx"]]
 245                compsep_run["tracers"] = initialize_scalar_tracers(config, input_fgds_alms, compsep_run, field=compsep_run["field"], **kwargs)
 246                del input_fgds_alms
 247            else:
 248                compsep_run["field_idx"] = 0
 249        output_maps = _ilc_scalar(config, input_alms, compsep_run, **kwargs)
 250    
 251    del compsep_run["field"]
 252    if compsep_run["method"] in ["mcilc", "mc_ilc", "mc_cilc"]:
 253        if "path_patches" not in compsep_run:
 254            del compsep_run["tracers"]
 255        else:
 256            compsep_run.pop("field_idx", None)
 257
 258    return output_maps

 294def _ilc_needlet(
 295    config: Configs,
 296    input_alms: np.ndarray,
 297    compsep_run: Dict,
 298    **kwargs: Any
 299) -> np.ndarray:
 300    """
 301    Perform ILC in the needlet domain.
 302
 303    Parameters
 304    ----------
 305        config : Configs
 306            Configuration object with global settings. See 'ilc' for details.
 307        input_alms : np.ndarray
 308            Input alm array. It must have shape (n_channels, n_alms, n_components).
 309        compsep_run : dict
 310            Component separation parameters including needlet configuration. See 'ilc' for details.
 311        **kwargs : dict
 312            Additional keyword arguments for hp.map2alm.
 313
 314    Returns
 315    -------
 316        np.ndarray
 317            Output maps after needlet domain ILC.
 318    """
 319    b_ell = _get_needlet_windows_(compsep_run["needlet_config"], config.lmax)
 320    if compsep_run["b_squared"]:
 321        b_ell = b_ell**2
 322
 323    if compsep_run['save_needlets']:
 324        compsep_run["path_out"] = _get_full_path_out(config, compsep_run)
 325        os.makedirs(compsep_run["path_out"], exist_ok=True)
 326        np.save(os.path.join(compsep_run["path_out"], "needlet_bands"), b_ell)
 327
 328    output_alms = np.zeros((input_alms.shape[1], input_alms.shape[-1]), dtype=complex)
 329    for j in range(b_ell.shape[0]):
 330        output_alms += _ilc_needlet_j(config, input_alms, compsep_run, b_ell[j], j, **kwargs)
 331    
 332    output_maps = np.array(
 333        [hp.alm2map(np.ascontiguousarray(output_alms[:, c]), config.nside, lmax=config.lmax, pol=False, pixwin=config.pixel_window_out)
 334            for c in range(input_alms.shape[-1])]).T
 335    
 336    if "mask" in compsep_run:
 337        output_maps[compsep_run["mask"] == 0.,:] = 0.
 338
 339    return output_maps

 422def _ilc_pixel(config: Configs, input_alms: np.ndarray, compsep_run: Dict, **kwargs) -> np.ndarray:
 423    """
 424    Computes ILC maps from input alms in pixel space using the specified component separation method.
 425
 426    Parameters
 427    ----------
 428        config : Configs 
 429            Configuration parameters. See 'ilc' for details.
 430        input_alms : np.ndarray
 431            Input spherical harmonic coefficients of shape (n_channels, n_alm, n_components).
 432        compsep_run : dict
 433            Dictionary specifying the component separation configuration. See 'ilc' for details.
 434        **kwargs : dict
 435            Additional keyword arguments for hp.map2alm.
 436
 437    Returns
 438    -------
 439        np.ndarray
 440            Reconstructed sky maps in pixel space.
 441    """
 442    npix = 12 * config.nside ** 2
 443    _, _, n_comps = input_alms.shape
 444    
 445    good_channels = _get_good_channels_nl(config, np.ones(config.lmax + 1))
 446    input_maps = np.zeros((good_channels.shape[0], npix, n_comps))
 447    
 448    for n, channel in enumerate(good_channels):
 449        input_maps[n] = np.array([
 450            hp.alm2map(np.ascontiguousarray(input_alms[channel, :, c]), config.nside,
 451                       lmax=config.lmax, pol=False) for c in range(n_comps)
 452        ]).T
 453
 454
 455    if compsep_run["method"]=="mcilc":
 456        if "path_patches" not in compsep_run:
 457            tracer = get_scalar_tracer(compsep_run["tracers"])
 458        else:
 459            tracer = None
 460        output_maps = _mcilc_maps(config, input_maps, tracer, compsep_run, np.ones(config.lmax+1), mask_mcilc=compsep_run.get("mask", None)) 
 461    else:
 462        output_maps = _ilc_maps(config, input_maps, compsep_run, np.ones(config.lmax+1), mask=compsep_run.get("mask", None))
 463    
 464    if config.pixel_window_out:
 465        for c in range(output_maps.shape[1]):
 466            alm_out = hp.map2alm(output_maps[:,c],lmax=config.lmax, pol=False, **kwargs)
 467            output_maps[:, c] = hp.alm2map(alm_out, config.nside, lmax=config.lmax,
 468                                           pol=False, pixwin=True)
 469
 470    if "mask" in compsep_run:
 471        output_maps[compsep_run["mask"] == 0.,:] = 0.
 472
 473    return output_maps

 475def _ilc_maps(config: Configs, input_maps: np.ndarray, compsep_run: Dict,
 476              b_ell: np.ndarray, nl_scale: Optional[Union[int, None]] = None, mask: Optional[np.ndarray] = None
 477            ) -> np.ndarray:
 478    """
 479    Computes ILC weights and outputs the cleaned map using ILC methods with inclusion of components deprojection (if needed).
 480
 481    Parameters
 482    ----------
 483        config : Configs
 484            Configuration object containing global settings. See 'ilc' for details.
 485        input_maps : np.ndarray
 486            Input maps of shape (n_channels, n_pixels, n_components).
 487        compsep_run : dict
 488            Dictionary with ILC run configuration. See 'ilc' for details.
 489        b_ell : np.ndarray
 490            Needlet harmonic window function. Used if `compsep_run["domain"]` is "needlet".
 491            If `compsep_run["domain"]` is "pixel", it should be an array of ones with shape (lmax+1,).
 492        nl_scale : int, optional
 493            Needlet scale index corresponding to the current ILC run. Used for:
 494            - Deprojection of components in cILC.
 495            - Save weights with proper label.
 496        mask : np.ndarray, optional
 497            HEALPix mask to exclude regions from covariance computation.
 498            It must be a 1D array with shape (12 * nside**2,). If non-binary, it will be used to weigh pixel in covariance computation.
 499
 500    Returns
 501    -------
 502        np.ndarray
 503            Cleaned output sky maps of shape (n_pixels, n_components).
 504    """
 505
 506    good_channels = _get_good_channels_nl(config, b_ell)
 507    freqs = np.array(config.instrument.frequency)[good_channels]
 508
 509    bandwidths = _get_bandwidths(config, good_channels)
 510
 511    if compsep_run["component_out"] == "cmb":
 512        A_out = _get_CMB_SED(freqs, units=config.units, bandwidths=bandwidths)
 513    elif compsep_run["component_out"] in ['0d', '1bd', '1Td', '2bd', '2Td', '2bdTd', '2Tdbd']:
 514        A_out = _get_SEDs(freqs, [compsep_run["component_out"]], beta_d=compsep_run["beta_d_out"], T_d=compsep_run["T_d_out"], nu_ref_d=compsep_run["nu_ref_d_out"], units=config.units, bandwidths=bandwidths)[0]
 515    elif compsep_run["component_out"] in ['0s', '1bs', '2bs']:
 516        A_out = _get_SEDs(freqs, [compsep_run["component_out"]], beta_s=compsep_run["beta_s_out"], nu_ref_s=compsep_run["nu_ref_s_out"], units=config.units, bandwidths=bandwidths)[0]
 517    else:
 518        A_out = _get_SEDs(freqs, [compsep_run["component_out"]], units=config.units, bandwidths=bandwidths)[0]
 519        
 520    if compsep_run["method"] == "cilc":
 521        if nl_scale is None:
 522            compsep_run["A"] = _get_SEDs(freqs, compsep_run["constraints"]["components"], beta_d=compsep_run["constraints"]["beta_d"], T_d=compsep_run["constraints"]["T_d"], beta_s=compsep_run["constraints"]["beta_s"], units=config.units, bandwidths=bandwidths)
 523            compsep_run["e"] = np.array(compsep_run["constraints"]["deprojection"])
 524        else:
 525            compsep_run["A"] = _get_SEDs(freqs, compsep_run["constraints"]["components"][nl_scale], beta_d=compsep_run["constraints"]["beta_d"][nl_scale], T_d=compsep_run["constraints"]["T_d"][nl_scale], beta_s=compsep_run["constraints"]["beta_s"][nl_scale], units=config.units, bandwidths=bandwidths)
 526            compsep_run["e"] = np.array(compsep_run["constraints"]["deprojection"][nl_scale])
 527
 528    elif (compsep_run["method"] == "c_ilc") and (nl_scale is not None) and (nl_scale in compsep_run["special_nls"]):
 529        compsep_run["A"] = _get_SEDs(freqs, compsep_run["constraints"]["components"][np.where(np.array(compsep_run["special_nls"]) == nl_scale)[0][0]], beta_d=compsep_run["constraints"]["beta_d"][np.where(np.array(compsep_run["special_nls"]) == nl_scale)[0][0]], T_d=compsep_run["constraints"]["T_d"][np.where(np.array(compsep_run["special_nls"]) == nl_scale)[0][0]], beta_s=compsep_run["constraints"]["beta_s"][np.where(np.array(compsep_run["special_nls"]) == nl_scale)[0][0]], units=config.units, bandwidths=bandwidths)
 530        compsep_run["e"] = np.array(compsep_run["constraints"]["deprojection"][np.where(np.array(compsep_run["special_nls"]) == nl_scale)[0][0]])
 531
 532    elif (compsep_run["method"] == "mc_cilc") and (nl_scale is not None) and (nl_scale not in compsep_run["special_nls"]):
 533        nl_scale_cilc = nl_scale-len(np.array(compsep_run["special_nls"])[np.array(compsep_run["special_nls"]) < nl_scale])
 534        compsep_run["A"] = _get_SEDs(freqs, compsep_run["constraints"]["components"][nl_scale_cilc], beta_d=compsep_run["constraints"]["beta_d"][nl_scale_cilc], T_d=compsep_run["constraints"]["T_d"][nl_scale_cilc], beta_s=compsep_run["constraints"]["beta_s"][nl_scale_cilc], units=config.units, bandwidths=bandwidths)
 535        compsep_run["e"] = np.array(compsep_run["constraints"]["deprojection"][nl_scale_cilc])
 536
 537    if compsep_run["from_splits"]:
 538        cov = get_ilc_cov(input_maps[...,0], config.lmax, compsep_run, config.fwhm_out, b_ell, mask=mask, input_maps_2=input_maps[...,1])
 539    else:
 540        cov = get_ilc_cov(input_maps[...,0], config.lmax, compsep_run, config.fwhm_out, b_ell, mask=mask)
 541
 542    if compsep_run["ilc_bias"] != 0. and mask is not None:
 543        mask_cov = (cov[0,0,:] != 0.).astype(float)
 544        if np.sum(mask_cov == 0.) == mask_cov.shape[0]:
 545            raise ValueError("The provided mask masks out all the pixels used for covariance computation.")
 546        elif np.sum(mask_cov == 0.) == 0:
 547            mask_cov = None
 548    else:
 549        mask_cov = None
 550
 551    if not compsep_run["from_splits"]:
 552        noise_debias = compsep_run["cov_noise_debias"] if compsep_run["domain"] == "pixel" else compsep_run["cov_noise_debias"][nl_scale]
 553        if noise_debias != 0.:
 554            if not ("load_noise_covariance" in compsep_run and compsep_run["load_noise_covariance"]):
 555                cov_n = get_ilc_cov(input_maps[...,compsep_run["noise_idx"]], config.lmax, compsep_run, config.fwhm_out, b_ell, mask=mask)
 556            else:
 557                path_noicov = _get_full_path_nuiscov(config, compsep_run)
 558                cov_n = (load_nuiscov(config, path_noicov, compsep_run,
 559                                    hp.npix2nside(input_maps.shape[-2]), "noise", nl_scale=nl_scale)).T
 560            cov = cov - noise_debias * cov_n
 561            del cov_n
 562
 563    if mask_cov is not None:
 564        cov[..., mask_cov == 0.] = np.repeat(np.mean(cov[..., mask_cov != 0.], axis=-1, keepdims=True), np.sum(mask_cov == 0.), axis=-1)
 565        mask_cov = None
 566    
 567    if mask_cov is not None:
 568        inv_cov = get_inv_cov(cov[..., mask_cov != 0.])
 569    else:
 570        inv_cov = get_inv_cov(cov)
 571    del cov
 572
 573    w_ilc = get_ilc_weights(A_out, inv_cov, input_maps.shape, compsep_run, mask_cov=mask_cov)
 574    del inv_cov
 575
 576#    if mask is not None and compsep_run["ilc_bias"] != 0.:
 577#        w_ilc[:, mask == 0.] = 0.
 578
 579    if compsep_run["save_weights"]:
 580#        if 'path_out' not in compsep_run:
 581        compsep_run["path_out"] = _get_full_path_out(config, compsep_run)
 582        save_ilc_weights(config, w_ilc, compsep_run,
 583                         hp.npix2nside(input_maps.shape[-2]), nl_scale=nl_scale)
 584    
 585    compsep_run.pop("A", None)
 586    compsep_run.pop("e", None)
 587
 588    if compsep_run["ilc_bias"] == 0.:
 589        output_maps = np.einsum('i,ijk->jk', w_ilc, input_maps)
 590    else:
 591        output_maps = np.einsum('ij,ijk->jk', w_ilc, input_maps)
 592
 593    return output_maps

 595def _mcilc_maps(config: Configs, input_maps: np.ndarray, tracer: Union[np.ndarray, None],
 596                compsep_run: Dict, b_ell: np.ndarray,
 597                nl_scale: Optional[Union[int, None]] = None, mask_mcilc: Optional[np.ndarray] = None) -> np.ndarray:
 598    """
 599    Computes foreground-cleaned sky maps using the MC-ILC method.
 600
 601    Depending on the specified `mc_type`, it delegates to either the CEA-based or 
 602    random partitioning (RP) implementation.
 603
 604    Parameters
 605    ----------
 606        config : Configs
 607            Configuration object containing instrument, resolution, and output settings. See 'ilc' for details.
 608        input_maps : np.ndarray
 609            Input sky maps of shape (n_channels, n_pixels, n_components).
 610        tracer : np.ndarray
 611            Tracer map used to define spatial partitions for independent component separation. 
 612            It should have shape (n_pixels,).
 613            If None, the function will attempt to load patches from compsep_run["path_patches"] if available; otherwise, it will raise an error.
 614        compsep_run : dict
 615            Dictionary specifying the MCILC method and associated parameters:
 616            - "mc_type": Type of MCILC ('cea' or 'rp').
 617            - "n_patches": Number of spatial patches.
 618            - Optional flags such as "save_weights".
 619            See 'ilc' for details.
 620        b_ell : np.ndarray
 621            Needlet harmonic window function. Shape is (lmax+1,).
 622            Used if `compsep_run["domain"]` is "needlet".
 623            If `compsep_run["domain"]` is "pixel", it should be an array of ones with shape (lmax+1,).
 624        nl_scale : int, optional
 625            Needlet scale index for the current ILC run. Used for:
 626            - saving weights with proper label.
 627        mask_mcilc : np.ndarray, optional
 628            HEALPix mask to exclude regions from covariance computation in MC-ILC.
 629            It must be a 1D array with shape (12 * nside**2,). If non-binary, it will be used to weigh pixel in covariance computation.
 630
 631    Returns
 632    -------
 633        np.ndarray
 634            Cleaned output sky maps of shape (n_pixels, n_components).
 635    """
 636    good_channels = _get_good_channels_nl(config, b_ell)
 637    bandwidths = _get_bandwidths(config, good_channels)
 638
 639    # Get CMB SED for the good channels
 640    A_cmb = _get_CMB_SED(np.array(config.instrument.frequency)[good_channels], units=config.units, bandwidths=bandwidths)
 641
 642    if tracer is None:
 643        return _mcilc_patches_(config, input_maps, compsep_run, A_cmb, nl_scale=nl_scale, mask_mcilc=mask_mcilc)
 644    else:
 645        if "cea" in compsep_run["mc_type"]:
 646            return _mcilc_cea_(config, input_maps, tracer, compsep_run, A_cmb, nl_scale=nl_scale, mask_mcilc=mask_mcilc)
 647        elif "rp" in compsep_run["mc_type"]:
 648            return _mcilc_rp_(config, input_maps, tracer, compsep_run, A_cmb, nl_scale=nl_scale, mask_mcilc=mask_mcilc)

 650def _mcilc_cea_(config: Configs, input_maps: np.ndarray, tracer: np.ndarray,
 651                compsep_run: Dict, A_cmb: np.ndarray,
 652                nl_scale: Optional[Union[int, None]] = None, mask_mcilc: Optional[np.ndarray] = None) -> np.ndarray:
 653    """
 654    Performs MC-ILC CMB reconstruction using CEA (Cluster of Equal Area) partitioning.
 655
 656    The sky is divided into non-overlapping spatial patches based on a tracer map and
 657    the Healpix grid, allowing region-specific ILC weight estimation.
 658
 659    Parameters
 660    ----------
 661        config : Configs
 662            Configuration object with all global settings. See 'ilc' for details.
 663        input_maps : np.ndarray
 664            Input sky maps with shape (n_channels, n_pixels, n_components).
 665        tracer : np.ndarray
 666            Tracer map that defines sky partition.
 667        compsep_run : dict
 668            Component separation configuration, including:
 669            - "n_patches": Number of CEA patches.
 670            - "save_weights": Flag to save weights.
 671            See 'ilc' for details.
 672        A_cmb : np.ndarray
 673            Spectral energy distribution (SED) vector for the CMB.
 674        nl_scale : int, optional
 675            Needlet scale index for the current ILC run. Used for:
 676            - saving weights with proper label.
 677        mask_mcilc : np.ndarray, optional
 678            HEALPix mask to exclude regions from covariance computation in MC-ILC.
 679            It must be a 1D array with shape (12 * nside**2,). If non-binary, it will be used to weigh pixel in covariance computation.
 680
 681    Returns
 682    -------
 683        np.ndarray
 684            Output sky map cleaned via region-specific ILC weights, shape (n_pixels, n_components).
 685    """
 686    if mask_mcilc is None:
 687        mask_mcilc = np.ones(input_maps.shape[-2])
 688
 689    patches = _cea_partition(tracer, compsep_run["n_patches"], mask=mask_mcilc)
 690    if compsep_run["save_patches"] and (compsep_run['nsim'] is None or int(compsep_run['nsim']) == config.nsim_start):
 691        #if 'path_out' not in compsep_run:
 692        compsep_run["path_out"] = _get_full_path_out(config, compsep_run)
 693        save_patches(config, patches, compsep_run, nl_scale=nl_scale)
 694
 695    if compsep_run["from_splits"]:
 696        w_mcilc = get_mcilc_weights(input_maps[...,0], patches, A_cmb, compsep_run, mask_mcilc=mask_mcilc, input_2=input_maps[...,1])
 697    else:
 698        w_mcilc = get_mcilc_weights(input_maps[...,0], patches, A_cmb, compsep_run, mask_mcilc=mask_mcilc)
 699
 700    if compsep_run["save_weights"]:
 701        #if 'path_out' not in compsep_run:
 702        compsep_run["path_out"] = _get_full_path_out(config, compsep_run)
 703        save_ilc_weights(config, w_mcilc, compsep_run, hp.npix2nside(input_maps.shape[-2]), nl_scale=nl_scale)
 704    
 705    compsep_run.pop("A", None)
 706    compsep_run.pop("e", None)
 707
 708    return np.einsum('ij,ijk->jk', w_mcilc, input_maps)

 710def _mcilc_rp_(config: Configs, input_maps: np.ndarray, tracer: np.ndarray,
 711               compsep_run: Dict, A_cmb: np.ndarray,
 712               iterations: int = 30, nl_scale: Optional[Union[int, None]] = None, mask_mcilc: Optional[np.ndarray] = None) -> np.ndarray:
 713    """
 714    Performs MCILC reconstruction using Random Partitioning (RP) of the sky.
 715
 716    The tracer map is used to generate multiple random spatial partitions. ILC weights
 717    are estimated independently for each realization and averaged across iterations
 718    to reduce bias and improve robustness.
 719
 720    Parameters
 721    ----------
 722        config : Configs
 723            Configuration object including instrument and output settings. See 'ilc' for details.
 724        input_maps : np.ndarray
 725            Input sky maps with shape (n_channels, n_pixels, n_components).
 726        tracer : np.ndarray
 727            Tracer map used to guide random partitions.
 728        compsep_run : dict
 729            Dictionary with MCILC-specific parameters:
 730            - "n_patches": Number of partitions per iteration.
 731            - "save_weights": Whether to save averaged ILC weights.
 732            See 'ilc' for details.
 733        A_cmb : np.ndarray
 734            Spectral energy distribution (SED) of the CMB.
 735        iterations : int, default=30
 736            Number of random partition iterations for averaging.
 737        nl_scale : int or str, optional
 738            Needlet scale index for the current ILC run. Used for:
 739            - saving weights with proper label.
 740        mask_mcilc : np.ndarray, optional
 741            HEALPix mask to exclude regions from covariance computation in MC-ILC.
 742            It must be a 1D array with shape (12 * nside**2,). If non-binary, it will be used to weigh pixel in covariance computation.
 743
 744    Returns
 745    -------
 746        np.ndarray
 747            Cleaned sky map averaged over random partition realizations,
 748            shape (n_pixels, n_components).
 749    """
 750    output_maps = np.zeros((input_maps.shape[1], input_maps.shape[-1]))
 751    if mask_mcilc is None:
 752        mask_mcilc = np.ones(input_maps.shape[-2])
 753
 754    do_save_patches = compsep_run["save_patches"] and (compsep_run['nsim'] is None or int(compsep_run['nsim']) == config.nsim_start)
 755
 756    if do_save_patches:
 757        patches_set = []
 758
 759    for it in range(iterations):  
 760        patches = _rp_partition(tracer, compsep_run["n_patches"], mask=mask_mcilc)
 761        if do_save_patches:
 762            patches_set.append(patches)
 763
 764        if compsep_run["from_splits"]:
 765            w_mcilc = get_mcilc_weights(input_maps[...,0], patches, A_cmb, compsep_run, mask_mcilc=mask_mcilc, input_2=input_maps[...,1])
 766        else:
 767            w_mcilc = get_mcilc_weights(input_maps[...,0], patches, A_cmb, compsep_run, mask_mcilc=mask_mcilc)
 768        if compsep_run["save_weights"]:
 769            if it == 0:
 770                w_mcilc_save = np.copy(w_mcilc) / iterations
 771            else:
 772                w_mcilc_save += w_mcilc / iterations
 773        output_maps += (np.einsum('ij,ijk->jk', w_mcilc, input_maps) / iterations)
 774        
 775    if do_save_patches:
 776        #if 'path_out' not in compsep_run:
 777        compsep_run["path_out"] = _get_full_path_out(config, compsep_run)
 778        save_patches(config, np.array(patches_set), compsep_run, nl_scale=nl_scale)
 779
 780    if compsep_run["save_weights"]:
 781        #if 'path_out' not in compsep_run:
 782        compsep_run["path_out"] = _get_full_path_out(config, compsep_run)
 783        save_ilc_weights(config, w_mcilc_save, compsep_run,
 784                         hp.npix2nside(input_maps.shape[-2]), nl_scale=nl_scale)
 785
 786    compsep_run.pop("A", None)
 787    compsep_run.pop("e", None)
 788    
 789    return output_maps

 791def _mcilc_patches_(config: Configs, input_maps: np.ndarray,
 792                compsep_run: Dict, A_cmb: np.ndarray,
 793                nl_scale: Optional[Union[int, None]] = None, mask_mcilc: Optional[np.ndarray] = None) -> np.ndarray:
 794    """
 795    Performs MC-ILC CMB reconstruction using patches loaded from disk (compsep_run["path_patches"]).
 796
 797    Parameters
 798    ----------
 799        config : Configs
 800            Configuration object with all global settings. See 'ilc' for details.
 801        input_maps : np.ndarray
 802            Input sky maps with shape (n_channels, n_pixels, n_components).
 803        compsep_run : dict
 804            Component separation configuration, including:
 805            - "path_patches": Path to load patches for MCILC.
 806            - "save_weights": Flag to save weights.
 807            See 'ilc' for details.
 808        A_cmb : np.ndarray
 809            Spectral energy distribution (SED) vector for the CMB.
 810        nl_scale : int, optional
 811            Needlet scale index for the current ILC run. Used for:
 812            - saving weights with proper label.
 813        mask_mcilc : np.ndarray, optional
 814            HEALPix mask to exclude regions from covariance computation in MC-ILC.
 815            It must be a 1D array with shape (12 * nside**2,). If non-binary, it will be used to weigh pixel in covariance computation.
 816
 817    Returns
 818    -------
 819        np.ndarray
 820            Output sky map cleaned via region-specific ILC weights, shape (n_pixels, n_components).
 821    """
 822    if mask_mcilc is None:
 823        mask_mcilc = np.ones(input_maps.shape[-2])
 824
 825    if (nl_scale is not None) and (not compsep_run["path_patches"].endswith(".npy")):
 826        filename_patches = compsep_run["path_patches"] + f"_nl{nl_scale}.npy"
 827        if not os.path.exists(filename_patches):
 828            nl_scale_upt = nl_scale
 829            while nl_scale_upt >= 0:
 830                filename_patches = compsep_run["path_patches"] + f"_nl{nl_scale_upt}.npy"
 831                if os.path.exists(filename_patches):
 832                    break
 833                else:
 834                    nl_scale_upt -= 1
 835            if nl_scale_upt < 0:
 836                print(f"Warning: No patches file found for any needlet scale up to {nl_scale}. Attempting to load from {compsep_run['path_patches']} without needlet scale suffix.")
 837                filename_patches = compsep_run["path_patches"] + f".npy"
 838            else:
 839                print(f"Warning: No patches file found for needlet scale {nl_scale}. Loaded patches file {filename_patches} for needlet scale {nl_scale_upt}.")
 840                
 841            if not os.path.exists(filename_patches):
 842                raise FileNotFoundError(f"No patches file found for needlet scales up to {nl_scale} at {compsep_run['path_patches']} with or without needlet scale suffix. Please ensure that the path is correct and the file exists.")
 843    else:
 844        filename_patches = compsep_run["path_patches"]
 845        if not filename_patches.endswith(".npy"):
 846            filename_patches += ".npy"
 847
 848    if not os.path.exists(filename_patches):
 849        raise FileNotFoundError(f"Patches file not found at {filename_patches}. Please ensure that the path is correct and the file exists.")   
 850        
 851    patches = np.load(filename_patches)
 852    patches = np.array(patches)
 853    if patches.ndim > 2:
 854        raise ValueError(f"Loaded patches have more than 2 dimension (shape {patches.shape}). Please ensure that the file at {filename_patches} contains a 1D or 2D array of patch indices.")
 855    if patches.ndim == 2:
 856        if patches.shape[0] < compsep_run["field_idx"] + 1:
 857            patches = patches[-1]
 858        else:
 859            patches = patches[compsep_run["field_idx"]]
 860
 861    try:
 862        nside_patches = hp.get_nside(patches)
 863    except TypeError:
 864        raise ValueError(f"Loaded patches do not appear to be in a valid HEALPix format. Please ensure that the file at {filename_patches} contains a 1D or 2D array of patch indices that can be interpreted as HEALPix maps.")
 865    
 866    if nside_patches < hp.npix2nside(input_maps.shape[-2]):
 867        patches = hp.ud_grade(patches, hp.npix2nside(input_maps.shape[-2]), order_in='RING', order_out='RING')
 868    elif nside_patches > hp.npix2nside(input_maps.shape[-2]):
 869        raise ValueError(f"Loaded patches have nside {nside_patches} which is greater than the nside of the input maps ({hp.npix2nside(input_maps.shape[-2])}). Please ensure that the patches file at {filename_patches} has nside less than or equal to that of the input maps. This may also depend on the 'adapt_nside' setting in compsep_run, so please check that as well.")
 870    
 871    if compsep_run["from_splits"]:
 872        w_mcilc = get_mcilc_weights(input_maps[...,0], patches, A_cmb, compsep_run, mask_mcilc=mask_mcilc, input_2=input_maps[...,1])
 873    else:

 886def get_ilc_weights(
 887    A_out: np.ndarray,
 888    inv_cov: np.ndarray,
 889    input_shapes: Tuple[int, ...],
 890    compsep_run: Dict,
 891    mask_cov: Optional[np.ndarray] = None
 892) -> np.ndarray:
 893    """
 894    Compute Internal Linear Combination (ILC) weights for a given spectral component (typically CMB).
 895
 896    Parameters
 897    ----------
 898        A_out : np.ndarray
 899            Spectral response vector of the target component (e.g., CMB), shape (n_channels,).
 900        inv_cov : np.ndarray
 901            Inverse of the covariance matrix. Can be 2D or 3D depending on bias settings.
 902        input_shapes : tuple
 903            Shape of the input map, used to reshape or broadcast weights.
 904        compsep_run : dict
 905            Dictionary specifying component separation settings, including:
 906            - 'A': Optional constraint matrix for deprojection of additional components. Shape (n_components, n_channels).
 907            - 'e': Deprojection vector. Shape (n_components,).
 908            - 'ilc_bias': Flag indicating if covariance is pixel-independent (0.0) or not.
 909        mask_cov : np.ndarray, optional
 910            Optional mask used in covariance computation, shape (n_pixels,). 
 911
 912    Returns
 913    -------
 914        np.ndarray
 915            ILC weights with shape (n_channels, n_pixels) if ilc_bias is not 0.0,
 916            otherwise (n_channels,).
 917    """
 918    if "A" in compsep_run:
 919        compsep_run["A"] = np.vstack((A_out, compsep_run["A"]))
 920        compsep_run["e"] = np.insert(compsep_run["e"], 0, 1.)
 921        if compsep_run["ilc_bias"] == 0.:
 922            inv_ACA = np.linalg.inv(
 923                np.einsum("zi,il->zl", compsep_run["A"], np.einsum("ij,lj->il", inv_cov, compsep_run["A"]))
 924            )
 925            w_ilc = np.einsum(
 926                "l,lj->j", compsep_run["e"],
 927                np.einsum("lz,zj->lj", inv_ACA, np.einsum("zi,ij->zj", compsep_run["A"], inv_cov))
 928            )
 929            del inv_ACA
 930        else:
 931            inv_ACA = np.linalg.inv(
 932                np.einsum("zi,ilk->zlk", compsep_run["A"], np.einsum("ijk,lj->ilk", inv_cov, compsep_run["A"])).T
 933            ).T
 934            w_ilc=np.zeros((input_shapes[0],input_shapes[-2]))
 935            w_ = np.einsum(
 936                "l,ljk->jk", compsep_run["e"],
 937                np.einsum("lzk,zjk->ljk", inv_ACA, np.einsum("zi,ijk->zjk", compsep_run["A"], inv_cov))
 938            )
 939            del inv_ACA
 940            for i in range(input_shapes[0]):
 941                if mask_cov is not None:
 942                    w__ = np.zeros(mask_cov.shape[0])
 943                    w__[mask_cov != 0.] = w_[i]
 944                    w_ilc[i] = hp.ud_grade(w__, hp.npix2nside(input_shapes[-2]))
 945                    del w__
 946                else:
 947                    w_ilc[i]=hp.ud_grade(w_[i],hp.npix2nside(input_shapes[-2]))
 948            del w_
 949    else:
 950        if compsep_run["ilc_bias"] == 0.:
 951            w_ilc = (A_out.T @ inv_cov) / (A_out.T @ inv_cov @ A_out) 
 952        else:
 953            w_ilc=np.zeros((input_shapes[0],input_shapes[-2]))
 954            AT_invC = np.einsum('j,ijk->ik', A_out, inv_cov) # np.sum(inv_cov,axis=1)
 955            AT_invC_A = np.einsum('j,ijk, i->k', A_out, inv_cov, A_out) #np.sum(inv_cov,axis=(0,1))
 956            for i in range(input_shapes[0]):
 957                if mask_cov is not None:
 958                    w__ = np.zeros(mask_cov.shape[0])
 959                    w__[mask_cov != 0.] = AT_invC[i]/AT_invC_A
 960                    w_ilc[i] = hp.ud_grade(w__, hp.npix2nside(input_shapes[-2]))
 961                    del w__
 962                else:
 963                    w_ilc[i]=hp.ud_grade(AT_invC[i]/AT_invC_A,hp.npix2nside(input_shapes[-2]))
 964    return w_ilc
