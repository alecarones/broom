 14def join(loader: yaml.Loader, node: yaml.Node) -> str:
 15    """
 16    Custom YAML tag to join sequences into a single string.
 17
 18    Parameters
 19    ----------
 20        loader : yaml.Loader
 21            The YAML loader instance.
 22        node : yaml.Node
 23            The YAML node to process.
 24
 25    Returns
 26    -------
 27        str
 28            A string formed by joining the sequence elements.
 29    """
 30    seq = loader.construct_sequence(node)
 31    return "".join([str(i) for i in seq])

185    def _generate_channel_tags(self):
186        self.channels_tags = []
187        unique_freqs, counts = np.unique(self.frequency, return_counts=True)
188        labels = [list(string.ascii_lowercase[:c]) for c in counts]
189
190        for freq in self.frequency:
191            idx = np.where(unique_freqs == freq)[0][0]
192            label = labels[idx].pop(0)
193            tag = f"{freq}GHz" if counts[idx] == 1 else f"{freq}{label}GHz"
194            self.channels_tags.append(tag)
