 579def get_gilc_maps(
 580    config: Configs,
 581    gilc_config: Dict[str, Any],
 582    nsim: Optional[str] = None
 583) -> SimpleNamespace:
 584    """
 585    Load GILC component separation results (total signal, noise residuals, and foreground residuals) of all frequency channels
 586    provided in the instrument object and for a given simulation run and field. 
 587
 588    Parameters
 589    -----------
 590        config : Configs
 591            Configuration object containing instrument and output specifications.
 592        gilc_config : Dict[str, Any]
 593            Dictionary containing GNILC configuration parameters, including:
 594                path_gilc : str
 595                    Root path to the GNILC output directory. The full path will be given by '{config.path_outputs}/{path_gilc}'.
 596                gilc_components : List[str]
 597                    List of GNILC components to load. Possible elements are "output_total", "noise_residuals", "fgds_residuals".
 598                field_in : Optional[str], default=None
 599                    Type of field to load ("T", "QU", "EB", "TQU", "TEB", etc.). If None, default is config.field_out.
 600        nsim : Optional[Union[str, int]], default=None
 601            Simulation identifier, if any (used to select specific simulation output files).
 602
 603    Returns
 604    --------
 605        gnilc_maps : SimpleNamespace
 606            A container of numpy arrays with attributes corresponding to the requested GNILC components:
 607    """
 608    path_gilc = gilc_config["gilc_path"]
 609    gilc_components = gilc_config["gilc_components"]
 610    nside = gilc_config["nside"]
 611    fwhm_out = gilc_config["fwhm_out"]
 612    lmax = gilc_config["lmax"]
 613    field_in = gilc_config["field_in"]
 614
 615    if not os.path.exists(os.path.join(config.path_outputs, path_gilc)):
 616        raise ValueError(f"Path {os.path.join(config.path_outputs, path_gilc)} does not exist.")
 617    if field_in is None:
 618        field_in = config.field_out
 619    
 620    gilc_maps = SimpleNamespace()
 621
 622    if field_in in ["TQU", "TEB"]:
 623        if config.field_out == "T":
 624            gilc_fields = 0
 625        elif config.field_out in ["QU", "QU_E", "QU_B", "E", "B"]:
 626            gilc_fields = (1,2)
 627        elif config.field_out in ["TQU","TEB"]:
 628            gilc_fields = (0,1,2)
 629    elif field_in in ["QU","EB"]:
 630        gilc_fields = (0,1)
 631    elif field_in in ["T","E","B"]:
 632        gilc_fields = 0
 633
 634    for component in gilc_components:
 635        if component.split("_")[-1] == "residuals":
 636            attr_out = component.split("_residuals")[0]
 637        elif "output" in component:
 638            attr_out = component.split("output_")[1]
 639
 640        filepath = os.path.join(config.path_outputs, path_gilc, component)
 641        if nsim is not None:
 642            filepath = os.path.join(filepath, nsim)
 643        if not os.path.exists(filepath):
 644            raise ValueError(f"Path {filepath} does not contain the expected GNILC {component} maps.")
 645
 646        setattr(gilc_maps, attr_out, [])
 647
 648        for f, freq in enumerate(config.instrument.frequency):
 649            tag = config.instrument.channels_tags[f]
 650            filename = f"{field_in}_{component}_{tag}_{fwhm_out}acm_ns{nside}_lmax{lmax}"
 651            if nsim is not None:
 652                filename += f"_{nsim}"
 653            filename += ".fits"
 654            getattr(gilc_maps, attr_out).append(hp.read_map(os.path.join(filepath, filename), field=gilc_fields))
 655        setattr(gilc_maps, attr_out, np.array(getattr(gilc_maps, attr_out)))
 656
 657    return gilc_maps
